## これなに

昔作った電気回路の電流を計算するC#のコードをPythonにしてみました。
[キルヒホッフの法則](https://ja.wikipedia.org/wiki/キルヒホッフの法則 (電気回路))を使っています。

## Pythonのコード

```py
from itertools import combinations
from typing import TYPE_CHECKING

import networkx as nx
from more_itertools import pairwise

if TYPE_CHECKING:
    from collections.abc import Hashable

    import pandas as pd


class CycleInfo:
    """サイクルの情報"""

    cycle: list[Hashable]
    volt: float  # サイクルの向きの起電力(他サイクルの影響込み)
    resist: float  # サイクル上の抵抗の和
    cur: float  # サイクル電流

    def __init__(self, g: nx.DiGraph, i: int, cycle: list[Hashable]) -> None:
        """g.edgesのcidsにインデックスを追加"""
        self.cycle = cycle
        self.volt = self.resist = self.cur = 0
        for e in pairwise([*cycle, cycle[0]]):
            # coe: 辺の向きがサイクルの向きと同じなら1、逆なら-1
            if dc := g.edges.get(e):
                coe = 1
            else:
                coe, dc = -1, g.edges[e[1], e[0]]
            self.volt += coe * dc["volt"]
            self.resist += dc["resist"]
            dc["cids"].append((coe, i))


def calc_current(df: pd.DataFrame) -> pd.DataFrame:
    """電流計算。結果はcur列に追加"""
    g = nx.DiGraph()
    for row in df.itertuples():
        g.add_edge(row.node1, row.node2, volt=row.volt, resist=row.resist, cids=[])

    cycles: list[list[Hashable]] = nx.cycle_basis(g.to_undirected())
    cinfos = [CycleInfo(g, i, cycle) for i, cycle in enumerate(cycles)]

    # vv[i][j]: サイクルiの電流1あたりにサイクルjの起電力が変わる量(共有辺の電圧降下).
    vv: list[list[float]] = [[0] * len(cinfos) for _ in cinfos]

    for dc in g.edges.values():
        for (coe1, cid1), (coe2, cid2) in combinations(dc["cids"], 2):
            vv[cid1][cid2] += -coe1 * coe2 * dc["resist"]
            vv[cid2][cid1] += -coe1 * coe2 * dc["resist"]

    # 収束するまで各サイクル電流を更新し、差分を他サイクルの起電力に反映
    modified = True
    while modified:
        modified = False
        for i, cinfo in enumerate(cinfos):
            if not cinfo.resist:
                if not cinfo.volt:
                    continue
                msg = "回路ショート"
                raise ValueError(msg)
            cur = cinfo.volt / cinfo.resist
            dif = cur - cinfo.cur
            if abs(dif) < 1e-8:
                continue
            modified = True
            cinfo.cur = cur
            for j, cinfo2 in enumerate(cinfos):
                cinfo2.volt += dif * vv[i][j]
    # 辺の電流は、その辺を含むサイクル電流の符号付き和(node1→node2が正)
    df["cur"] = [
        sum(coe * cinfos[i].cur for coe, i in g.edges[r.node1, r.node2]["cids"])
        for r in df.itertuples()
    ]
    return df

```

## 追記

サイクルを使わず、また、連立方程式を解く方法。

```python
from itertools import islice

import networkx as nx
import numpy as np


def calc_flow(g: nx.DiGraph) -> None:
    """キルヒホッフの法則で電流を計算。結果はflowに追加"""
    num_nodes = g.number_of_nodes()
    num_edges = g.number_of_edges()
    node_idx = {node: i for i, node in enumerate(g.nodes)}
    size = num_nodes + num_edges
    a = np.zeros((size, size))
    b = np.zeros(size)
    a[0, 0] = 1
    for i, node in enumerate(islice(g.nodes, 1, num_nodes), 1):
        for j, (u, v) in enumerate(g.edges):
            if u == node:
                a[i, num_nodes + j] = 1
            if v == node:
                a[i, num_nodes + j] = -1

    # V_u - V_v + V_src = I * R  =>  V_u - V_v - I * R = -V_src
    for j, (u, v, data) in enumerate(g.edges(data=True)):
        r = data.get("resist", 0)
        v_src = data.get("volt", 0)
        row = num_nodes + j
        a[row, node_idx[u]] = 1
        a[row, node_idx[v]] = -1
        a[row, num_nodes + j] = -r
        b[row] = -v_src

    x = np.linalg.solve(a, b)
    for j, (u, v) in enumerate(g.edges):
        g[u][v]["flow"] = x[num_nodes + j]


if __name__ == "__main__":
    g = nx.DiGraph()
    g.add_edge("0", "1", volt=0, resist=20)
    g.add_edge("0", "2", volt=0, resist=10)
    g.add_edge("1", "2", volt=0, resist=20)
    g.add_edge("1", "3", volt=0, resist=50)
    g.add_edge("2", "3", volt=0, resist=40)
    g.add_edge("3", "4", volt=0, resist=0)
    g.add_edge("4", "5", volt=30, resist=0.98)
    g.add_edge("5", "0", volt=0, resist=0)
    calc_flow(g)
    for u, v, data in g.edges(data=True):
        print(f"Edge ({u}, {v}): flow={data.get('flow', 'N/A'):.4f} A")
```
